\begin{finalpart}
\section{Reproducibility}
\label{sec:repro}
Every table and figure of the final-phase sections can be regenerated from files in the archive without re-running the board; re-running the board needs the Boolean Board and the bitstreams in \path{fpga_implementation/v6/build*/}.

\noindent\textbf{File map.}
\begin{itemize}
    \item \textbf{Algorithm and models:} \path{fpga_implementation/v6/model/} (C model \texttt{fpga\_model.c}, PC brain \texttt{brain.c}, look-ahead model \texttt{v7\_post.c}); Python prototypes and studies in \path{tools/accuracy/}.
    \item \textbf{RTL and builds:} \path{fpga_implementation/v6/rtl/}, testbenches and regression in \path{sim/}, Vivado scripts \texttt{build\_v7.tcl} / \texttt{program\_v7.tcl}, reports \texttt{build/\{,v61/,v7/\}\{util,util\_hier,timing,power\}.rpt}.
    \item \textbf{Per-record results:} board runs \path{report/rtl_pc_split/board_umem/} (v6), \path{report/v61_v7/board_v61/} (v6.1), \path{report/v61_v7/board_v7/} (v7-DL); C-model and study results \path{tools/accuracy/results_*.csv}; mid-term results \path{code/results/}.
    \item \textbf{Report tools:} \path{report/final_report/tools/}, \path{tools/report_stats/} (outputs in \path{tools/report_stats/out/}).
\end{itemize}

\noindent\textbf{Commands} (from the repository root, Python in \texttt{venv}):
\begin{itemize}
    \item Board runs and bit-exact check: \path{V6_VARIANT=v7 python fpga_implementation/uart_host/run_v6_board_eval.py --sets cdev,ad,ctest,syn} (\texttt{v61} for v6.1) $\rightarrow$ Tabs.~\ref{tab:final_results}, \ref{tab:v7_results}.
    \item \path{python report/final_report/tools/make_final_figures.py} $\rightarrow$ result figures \texttt{fin\_*.png} of v6.
    \item \path{python report/final_report/tools/final_analysis.py} $\rightarrow$ Tab.~\ref{tab:final_ci}, failure analysis (\path{tables/cinc_failures.csv}).
    \item \path{python report/final_report/tools/make_v7_figures.py} $\rightarrow$ \texttt{fin\_v7\_*.png}, \path{tables/board_v7_f1_ci.csv} (Tab.~\ref{tab:v7_results}).
    \item \path{bash fpga_implementation/v6/sim/wave/run_wave.sh}, then \path{python report/final_report/tools/plot_waves.py} $\rightarrow$ simulation figures \texttt{wave\_*.png}.
    \item \path{python report/final_report/tools/fix_midterm_figures.py} $\rightarrow$ corrected mid-term figures.
    \item \path{python report/final_report/tools/pc_link_traffic.py} $\rightarrow$ Tab.~\ref{tab:pc_traffic}.
    \item \path{python tools/report_stats/stats.py} $\rightarrow$ Sec.~\ref{sec:stats}, Tab.~\ref{tab:paired}, clustered CIs, shortfall.
    \item \path{python report/final_report/tools/check_final_numbers.py} $\rightarrow$ checks that the report quotes the numbers in the result files.
    \item \path{python tools/report_stats/checksums.py} $\rightarrow$ checksums.
\end{itemize}

\noindent\textbf{Checksums.} SHA-256 checksums of the 53 result files and Vivado reports used by the report are in \path{tools/report_stats/out/sha256.csv}. % source: tools/report_stats/checksums.py output (42 files, 0 missing)
